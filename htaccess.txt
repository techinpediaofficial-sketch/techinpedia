# Override default cPanel error pages to prevent directfwd malware/parking script injections
ErrorDocument 400 "Bad Request"
ErrorDocument 401 "Unauthorized"
ErrorDocument 403 "Forbidden"
ErrorDocument 404 "Not Found"
ErrorDocument 500 "Internal Server Error"
ErrorDocument 502 "Bad Gateway"
ErrorDocument 503 "Service Unavailable"

<IfModule mod_headers.c>
  Header always set Content-Security-Policy "upgrade-insecure-requests;"
  RequestHeader set X-Forwarded-Proto "https"
  RequestHeader set X-Forwarded-Port "443"
</IfModule>

RewriteEngine On

RewriteCond %{HTTP_HOST} ^www\.techinpedia\.com$ [NC]
RewriteRule ^(.*)$ https://techinpedia.com/$1 [R=301,L]

RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://techinpedia.com/$1 [R=301,L]

RewriteCond %{REQUEST_URI} ^/\.well-known/acme-challenge/
RewriteRule .* - [L]

RewriteCond %{HTTP:Upgrade} =websocket [NC]
RewriteRule ^(.*)$ ws://127.0.0.1:3000/$1 [P,L]

RewriteRule ^(.*)$ http://127.0.0.1:3000/$1 [P,L,QSA]
